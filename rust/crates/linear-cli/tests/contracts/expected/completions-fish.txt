# Walk the words before the cursor through the command tree and print the
# selected command path. Option values are skipped, `--` ends command
# selection, and an unknown word under a parent selects nothing.
function __fish_linear_command_path
    set -l words (commandline -opc)
    set -e words[1]
    set -l path 'linear'
    set -l value_next 0
    for word in $words
        if test $value_next -eq 1
            set value_next 0
            continue
        end
        switch $word
            case '--'
                return 1
            case '--*=*'
                set -l option (string replace -r '=.*$' '' -- $word)
                __fish_linear_known_option $path $option
                or return 1
            case '--*'
                __fish_linear_known_option $path $word
                or return 1
                __fish_linear_takes_value $path $word
                and set value_next 1
            case '-?*'
                # A short cluster: a valued letter takes the rest of the word,
                # or the next word when it is last.
                set -l letters (string sub --start 2 -- $word | string split '')
                while set -q letters[1]
                    set -l letter $letters[1]
                    set -e letters[1]
                    __fish_linear_known_option $path -$letter
                    or return 1
                    if __fish_linear_takes_value $path -$letter
                        set -q letters[1]
                        or set value_next 1
                        break
                    end
                end
            case '*'
                contains -- $path 'linear' 'linear auth' 'linear auth help' 'linear issue' 'linear issue comment' 'linear issue comment help' 'linear issue relation' 'linear issue relation help' 'linear issue agent-session' 'linear issue agent-session help' 'linear issue help' 'linear issue help comment' 'linear issue help relation' 'linear issue help agent-session' 'linear team' 'linear team help' 'linear user' 'linear user help' 'linear project' 'linear project comment' 'linear project comment help' 'linear project help' 'linear project help comment' 'linear project-update' 'linear project-update help' 'linear cycle' 'linear cycle help' 'linear milestone' 'linear milestone help' 'linear initiative' 'linear initiative comment' 'linear initiative comment help' 'linear initiative help' 'linear initiative help comment' 'linear initiative-update' 'linear initiative-update help' 'linear label' 'linear label help' 'linear template' 'linear template help' 'linear document' 'linear document comment' 'linear document comment help' 'linear document help' 'linear document help comment' 'linear completions' 'linear completions help' 'linear help' 'linear help auth' 'linear help issue' 'linear help issue comment' 'linear help issue relation' 'linear help issue agent-session' 'linear help team' 'linear help user' 'linear help project' 'linear help project comment' 'linear help project-update' 'linear help cycle' 'linear help milestone' 'linear help initiative' 'linear help initiative comment' 'linear help initiative-update' 'linear help label' 'linear help template' 'linear help document' 'linear help document comment' 'linear help completions'
                or continue
                set path (__fish_linear_subcommand $path $word)
                or return 1
        end
    end
    echo $path
end

# Succeed when the words before the cursor select exactly the path $argv[1].
# Every entry asks, so the walk is cached per escaped word list.
function __fish_linear_using_command
    set -l key (commandline -opc | string escape | string join ' ')
    if not set -q __fish_linear_command_key; or test "$key" != "$__fish_linear_command_key"
        set -g __fish_linear_command_key $key
        set -g __fish_linear_command (__fish_linear_command_path)
    end
    test "$__fish_linear_command" = "$argv[1]"
end
